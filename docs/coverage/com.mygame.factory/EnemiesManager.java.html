package com.mygame.factory;

import com.badlogic.gdx.graphics.glutils.ShapeRenderer;
import com.badlogic.gdx.physics.box2d.World;
import com.badlogic.gdx.utils.Array;
import com.mygame.entities.Enemy.Enemy;
import com.mygame.entities.Enemy.Military.Military;
import com.mygame.entities.Enemy.Zombie.Zombie;
import com.mygame.entities.Player.Player;
import com.mygame.items.Weapons.Weapon;

public class EnemiesManager {

    private final Array<Enemy> enemies = new Array<>();
    // spawn enemies 
    public Zombie spawnZombie(World world, float x, float y) {

        Zombie zombie = new Zombie(world, x, y);

        enemies.add(zombie);

        return zombie;
    }

    public Military spawnMilitary(World world, float x, float y, Weapon weapon, BulletManager bulletFactory){
        Military military = new Military(world, x, y, weapon, bulletFactory);
        enemies.add(military);
        return military;
    }

    public void update(Player player, float delta) {

        for (Enemy enemy : enemies) {
            if (enemy.isDead()) {
                enemy.updateDeath(delta);
                continue;
            }
            enemy.update(player, delta);
        }
    }

    // draw enemies
    public void draw(ShapeRenderer shapes) {

        for (Enemy enemy : enemies) {
            enemy.draw(shapes);
        }
    }

    public void removeDeadEnemies(World world) {
        for (int i = enemies.size - 1; i >= 0; i--) {
            Enemy enemy = enemies.get(i);

            if (enemy.canBeDisable()) {
                enemy.disableCollision();
                enemies.removeIndex(i);
            }
        }
    }   

    public Array<Enemy> getEnemies() {
        return enemies;
    }
}
